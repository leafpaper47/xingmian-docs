## 注册指令（plugin.yml）
/mythicmobs | usage=None | desc='MythicMobs base command' | alias=['mm'] | perm=None
/mythicmobsmenu | usage=None | desc='MythicMobs menu command' | alias=['mmmenu', 'mmm'] | perm=None
/spawnmob | usage=None | desc='Spawn a mob' | alias=['spawnmythicmob', 'mspawn', 'msummon'] | perm=None
/pins | usage=None | desc='Easier command for pins' | alias=['pin'] | perm=None
/addpin | usage=None | desc='quickly adds a pin' | alias=None | perm=None
