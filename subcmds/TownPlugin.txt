# TownPlugin  (TownPlugin-1.4.8-itemguard.jar)
## 注册指令
  /town  usage=None desc='城镇系统（打开城镇核心界面）' alias=['tw'] perm=None
  /townui  usage=None desc='城镇系统（打开创建界面）' alias=['twu'] perm=None
  /townadmin  usage=None desc='城镇系统管理面板（需 town.admin 权限）' alias=['twa'] perm=None
  /townwarp  usage=None desc='城镇传送（全服玩家可查看/传送至各城镇传送点）' alias=['twarp'] perm=None
  /wh  usage=None desc='城镇仓库（打开所属城镇的共享仓库，需先加入城镇）' alias=None perm=None

## 代码类
  cn.yxsjmc.town.CommandHandler
  cn.yxsjmc.town.warehouse.WarehouseCommand

## javap 字符串 token
  accept delete deny town townadmin townui townwarp warehouse
