# OpenInv  (OpenInv.jar)
## 注册指令
  /openinv  usage="/<command> [Player] - Open a player's inventory" desc="Open a player's inventory" alias=['oi', 'inv', 'open'] perm='openinv.inventory.open.self;openinv.inventory.open.other'
  /clearinv  usage="/<command> [Player] - Clear a player's inventory" desc="Clear a player's inventory" alias=None perm='openinv.clear.self;openinv.clear.other'
  /openender  usage="/<command> [Player] - Open a player's ender chest" desc="Open a player's ender chest" alias=['oe'] perm='openinv.enderchest.open.self;openinv.enderchest.open.other'
  /clearender  usage="/<command> [Player] - Clear a player's ender chest" desc="Clear a player's ender chest" alias=None perm='openinv.clear.self;openinv.clear.other'
  /searchinv  usage='/<command> <Material> [MinAmount] - MinAmount is optional, the minimum amount required' desc='Search and list players having a specific item' alias=['si'] perm='openinv.search.inventory'
  /searchender  usage='/<command> <Material> [MinAmount] - MinAmount is optional, the minimum amount required' desc='Search and list players having a specific item in their ender chest' alias=['se'] perm='openinv.search.inventory'
  /silentcontainer  usage='/<command> [check|on|off] - Check, toggle, or set SilentContainer' desc='SilentContainer stops sounds and animations when using containers.' alias=['sc', 'silent', 'silentchest'] perm='openinv.container.silent'
  /anycontainer  usage='/<command> [check|on|off] - Check, toggle, or set AnyContainer' desc='AnyContainer allows using blocked containers.' alias=['ac', 'anychest'] perm='openinv.container.any'
  /searchenchant  usage='/<command> <[Enchantment] [MinLevel]> - Enchantment is the enchantment type, MinLevel is the minimum level. One is optional' desc='Search and list players with a specific enchantment.' alias=['searchenchants'] perm='openinv.search.inventory'
  /searchcontainer  usage='/<command> <Material> [ChunkRadius] - ChunkRadius is optional, the length that will be searched for matching items. Default 5' desc='Search and list containers with a specific material.' alias=['searchchest'] perm='openinv.search.container'
