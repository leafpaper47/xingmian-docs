# LeafHat  (LeafHat-1.0.3.jar)
## 注册指令
  /hat  usage='/hat [remove]' desc='戴上或脱下帽子' alias=['wearhat'] perm=None
  /unhat  usage='/unhat' desc='脱下头上的物品' alias=None perm=None

## 代码类
  me.leafhat.HatCommand
  me.leafhat.PluginCommands

## javap 字符串 token
  craftengine customid hat instance off reload rem remove unhat wear
