# LeafChat  (leafchat-1.0.5.jar)
## 注册指令
  /chatcolor  usage='/chatcolor [颜色ID] | /chatcolor gradient <起色> <止色>' desc='设置或打开聊天颜色选择' alias=None perm='leafchat.chatcolor'
  /msg  usage='/msg <玩家> <消息>' desc='发送私聊消息' alias=None perm='leafchat.msg'
  /r  usage='/r <消息>' desc='回复上次私聊对象' alias=None perm='leafchat.msg'
  /spymsg  usage='/spymsg' desc='切换管理员私聊监听' alias=None perm='leafchat.spy'
  /mute  usage='/mute <玩家> [时长]' desc='禁言玩家（留空时长=永久）' alias=None perm='leafchat.mute'
  /unmute  usage='/unmute <玩家>' desc='解除玩家禁言' alias=None perm='leafchat.unmute'
  /leafchat  usage='/leafchat reload' desc='LeafChat 主命令（重载配置等）' alias=None perm=None
  /leafchatview  usage='/leafchatview' desc='聊天物品展示后端（由 [inv]/[ec] 链接触发，无需手动使用）' alias=None perm=None
