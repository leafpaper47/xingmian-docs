  · # 可用的按钮名：
  · #   filler        边框玻璃

  · #   back          返回

  · #   next_page     下一页
  · #   create        新建
  · #   save          保存
  · #   delete        删除
  · #   add           添加
  · #   count         数量
  · #   mode          模式
  · #   rotation      研磨圈数
  · #   liquid        汤底 / 液体
  · #   carrier_none  空手盛出
  · #   quality_hint  成品左侧悬停品质提示（森罗图标缺失时回退为黄色染料）
  · # 示例：
  · # 厨具的图标与显示名。两项都可以只写其中一个。
  · # 可用的厨具名：pot、stockpot、steamer、shawarma、millstone、chopping_board、teapot
  · # 示例：
  · #     name: "铁锅"
  · #     name: "磨盘"
  · # 界面标题。支持 MiniMessage 标签（如 <gold>、<bold>）。
  · # 占位符：<appliance> 厨具名、<count> 条目数、<recipe> 成品物品名
  · # 不写颜色标签时保持内置的深灰色。
  · # 可用的界面名与默认值：
  · #   home_browse       食谱一览 - 选择厨具
  · #   home_edit         食谱编辑 - 选择厨具
  · #   list_browse       一览 - <appliance> (<count>)
  · #   list_edit         编辑 - <appliance> (<count>)
  · #   create_pick_type  新建食谱 - <appliance>
  · #   detail_accurate   精准食谱 - <recipe>
  · #   detail_flex       模糊食谱 - <recipe>
  · #   detail_chopping   砧板食谱 - <recipe>
  · #   detail_teapot     茶壶食谱 - <recipe>
  · #   soup_base         汤底表 - 共 <count> 种
  · # 示例：
  · #   home_browse: "<gold><bold>菜谱大全"
  · #   list_browse: "<yellow><appliance></yellow> 共 <count> 道"
