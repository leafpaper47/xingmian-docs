# WineSuite  (WineSuite-1.2.0.jar)
## 注册指令
  /winebook  usage='/winebook [reload]' desc='打开酿酒图鉴' alias=['酒鉴', '酿酒图鉴', '图鉴', 'wbk'] perm=None
  /winebuyer  usage='/winebuyer [list|reload|reset]' desc='打开酒水收购界面' alias=['wb', '收酒', '酒收'] perm=None
