# GSit  (GSit-3.5.1.jar)
## 注册指令
  /gsit  usage='/<command> [toggle|playertoggle]' desc='Sit command' alias=['sit'] perm=None
  /glay  usage='/<command>' desc='Lay command' alias=['lay'] perm='GSit.Lay'
  /glayback  usage='/<command>' desc='Lay back command' alias=['layback'] perm='GSit.LayBack'
  /gbellyflop  usage='/<command>' desc='Bellyflop command' alias=['bellyflop'] perm='GSit.Bellyflop'
  /gspin  usage='/<command>' desc='Spin command' alias=['spin'] perm='GSit.Spin'
  /gcrawl  usage='/<command> [toggle]' desc='Crawl command' alias=['crawl'] perm=None
  /gsitreload  usage='/<command>' desc='Plugin reload command' alias=['gsitrl'] perm='GSit.Reload'
